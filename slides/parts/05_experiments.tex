\begin{frame}{4.3\quad Kết hợp T2L + GEPA}{Thiết lập GEPA}

  \renewcommand{\arraystretch}{1.35}
  \begin{tabular}{@{}>{\bfseries}l>{\raggedright\arraybackslash}p{0.7\linewidth}@{}}

    Phạm vi sửa     & seed = luật của chính task; 3 ví dụ mẫu cố định, GEPA chỉ sửa \textbf{instruction}\\
    Đánh giá cuối   & instruction tốt nhất chạy lại trên 1.000 câu test\\
    \bottomrule
  \end{tabular}

% ------------------------------------------------------------------ conclusion
\begin{frame}{4.3\quad Kết hợp T2L + GEPA}{Kết luận}
  \footnotesize
  \begin{enumerate}\setlength{\itemsep}{0.55em}
    \item \textbf{Tái lập được xu hướng của T2L}, mức tăng nhỏ hơn paper vì base của harness
          mạnh hơn; số chính là T2L + ICL3 (+4,8 pp).
    \item \textbf{T2L mạnh hơn và rẻ hơn GEPA} trên cùng 5 dataset; lợi ích của GEPA chủ yếu
          đến từ ví dụ mẫu.
    \item \textbf{Hai phương pháp sửa hai kiểu lỗi khác nhau}: adapter thêm năng lực, prompt
          sửa định dạng.
    \item \textbf{Chỉ kết hợp được khi tìm prompt \emph{cùng} adapter.} Ghép prompt của base
          làm mất 4--6 pp; đồng tối ưu cho +0,1 đến +1,1 pp, cần nhiều seed để khẳng định.
  \end{enumerate}
\end{frame}
